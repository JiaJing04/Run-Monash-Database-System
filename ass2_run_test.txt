SQL> CREATE OR REPLACE PROCEDURE prc_entry_registration(
  2      p_competitor_no   IN NUMBER,
  3      p_carnival_name   IN VARCHAR2,
  4      p_event_type_desc IN VARCHAR2,
  5      p_team_name       IN VARCHAR2,
  6      p_charity_name    IN VARCHAR2,
  7      p_output          OUT VARCHAR2
  8  ) IS
  9      var_carnival_date   DATE;
 10      var_event_id        NUMBER;
 11      var_team_id         NUMBER;
 12      var_charity_id      NUMBER;
 13      var_entry_found     NUMBER;
 14      var_team_found      NUMBER;
 15      var_entry_no        NUMBER;
 16      var_charity_found   NUMBER;
 17  BEGIN
 18      IF (p_competitor_no IS NULL OR p_carnival_name IS NULL OR p_event_type_desc is NULL) THEN
 19          p_output := 'Incomplete registration information (competitor number, carnival name, event type), registration process cancelled';
 20      ELSE
 21          -- one competitor can only enter one event at a particular carnival
 22          SELECT COUNT(*) INTO var_entry_found
 23          FROM entry e JOIN event ev ON e.event_id = ev.event_id JOIN carnival c ON ev.carn_date = c.carn_date
 24          WHERE e.comp_no = p_competitor_no AND UPPER(c.carn_name) = UPPER(p_carnival_name);
 25  
 26          -- when competitor join 1 event already, it shouldnt join another one
 27          IF var_entry_found > 0 THEN
 28              p_output := 'Invalid multiple entry, registration process cancelled';
 29          ELSE
 30              -- get the carnival date
 31              SELECT c.carn_date INTO var_carnival_date
 32              FROM carnival c
 33              WHERE UPPER(c.carn_name) = UPPER(p_carnival_name);
 34  
 35              -- get the event id
 36              SELECT e.event_id INTO var_event_id
 37              FROM event e JOIN eventtype et ON e.eventtype_code = et.eventtype_code
 38              WHERE to_char(e.carn_date, 'DD/MON/YYYY') = to_char(var_carnival_date, 'DD/MON/YYYY') AND UPPER(et.eventtype_desc) = UPPER(p_event_type_desc);
 39  
 40              -- get new entry no
 41              SELECT COUNT(*) + 1 INTO var_entry_no
 42              FROM entry e
 43              WHERE e.event_id = var_event_id;
 44  
 45              -- check if team exists
 46              SELECT COUNT(*) INTO var_team_found
 47              FROM team t JOIN carnival c ON t.carn_date = c.carn_date
 48              WHERE UPPER(t.team_name) = UPPER(p_team_name) AND to_char(t.carn_date, 'DD/MON/YYYY') = to_char(var_carnival_date, 'DD/MON/YYYY');
 49  
 50              -- check if charity exists
 51              SELECT COUNT(*) INTO var_charity_found
 52              FROM charity c
 53              WHERE UPPER(c.char_name) = UPPER(p_charity_name);
 54  
 55              -- check if charity exists
 56              IF (var_charity_found > 0 OR p_charity_name is NULL) THEN
 57  
 58                  IF p_charity_name is NOT NULL THEN
 59                      SELECT c.char_id INTO var_charity_id
 60                      FROM charity c
 61                      WHERE UPPER(c.char_name) = UPPER(p_charity_name);
 62                  END IF;
 63  
 64                  -- insert entry
 65                  INSERT INTO entry (
 66                      event_id,
 67                      entry_no,
 68                      entry_starttime,
 69                      entry_finishtime,
 70                      entry_elapsedtime,
 71                      comp_no,
 72                      team_id
 73                  ) VALUES ( 
 74                      var_event_id,
 75                      var_entry_no,
 76                      NULL,
 77                      NULL,
 78                      NULL,
 79                      p_competitor_no,
 80                      NULL
 81                  );
 82  
 83                  IF p_charity_name is NOT NULL THEN
 84                      INSERT INTO entry_charity (
 85                          event_id, 
 86                          entry_no, 
 87                          char_id, 
 88                          char_percentage
 89                      ) VALUES ( 
 90                          var_event_id,
 91                          var_entry_no,
 92                          var_charity_id,
 93                          100
 94                      );
 95                  END IF;
 96
